class RecoveryClient(u.GuardedIBClient):
 def __init__(self,raw): super().__init__(raw); self.sent=False
 def placeOrder(self,contract,order):
  self.require_du()
  if self.sent or int(contract.conId)!=11470910 or contract.symbol!='BOQ' or order.action!='SELL' or order.orderType!='STP' or order.tif!='GTC' or Decimal(str(order.totalQuantity))!=13586 or Decimal(str(order.auxPrice))!=Decimal('6.07') or order.account!='DUQ200898': raise ValueError('unapproved recovery submission')
  self.sent=True
  return self.client.placeOrder(contract,order)
 def cancelOrder(self,*a,**k): raise ValueError('no further cancellation authorised')
async def main():
 s=Settings();u.require_settings(s)
 raw=IB();client=RecoveryClient(raw);bus=EventBus();adapter=IBAdapter(client,bus,settings=s,read_only=False)
 guard=u.ApprovedBroker(adapter)
 report={'status':'halted','scope':'single pinned BOQ protection recovery only','started_sydney':datetime.now(u.SYDNEY).isoformat()}

asyncio.run(main())
